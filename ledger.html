🗄️ Supabase 建表 SQL 指令

create table if not exists public.ledger_records (
  id bigint generated by default as identity primary key,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  tx_date date not null,
  tx_type text not null,
  account text not null,
  amount integer not null,
  card_qty integer default 0,
  points_added integer default 0,
  operator text default '系統',
  note text
);

alter table public.ledger_records enable row level security;
drop policy if exists "Allow public access to ledger_records" on public.ledger_records;
create policy "Allow public access to ledger_records"
  on public.ledger_records for all
  using (true)
  with check (true);
